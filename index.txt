    // 获取天气数据（OpenWeatherMap）
    async function fetchWeatherData(city) {
      try {
        const currentUrl = `https://api.openweathermap.org/data/2.5/weather?lat=${city.lat}&lon=${city.lon}&appid=${OPENWEATHER_API_KEY}&units=metric&lang=zh_cn`;
        const currentRes = await fetch(currentUrl);
        const currentData = await currentRes.json();
        
        const forecastUrl = `https://api.openweathermap.org/data/2.5/forecast?lat=${city.lat}&lon=${city.lon}&appid=${OPENWEATHER_API_KEY}&units=metric&lang=zh_cn`;
        const forecastRes = await fetch(forecastUrl);
        const forecastData = await forecastRes.json();
        
        const dailyMap = {};
        const hourlyData = { relative_humidity_2m: [], surface_pressure: [], visibility: [] };
        
        for (const item of forecastData.list) {
          const date = item.dt_txt.split(' ')[0];
          if (!dailyMap[date]) {
            dailyMap[date] = { temps: [], weatherCodes: [], precip: 0, windSpeeds: [], windDirs: [], humidities: [], pressures: [], visibilities: [] };
          }
          dailyMap[date].temps.push(item.main.temp);
          dailyMap[date].weatherCodes.push(item.weather[0].id);
          dailyMap[date].precip += (item.rain ? (item.rain['3h'] || 0) : 0);
          dailyMap[date].windSpeeds.push(item.wind.speed);
          dailyMap[date].windDirs.push(item.wind.deg);
          dailyMap[date].humidities.push(item.main.humidity);
          dailyMap[date].pressures.push(item.main.pressure);
          dailyMap[date].visibilities.push(item.visibility);
          hourlyData.relative_humidity_2m.push(item.main.humidity);
          hourlyData.surface_pressure.push(item.main.pressure);
          hourlyData.visibility.push(item.visibility);
        }
        
        const dates = Object.keys(dailyMap).sort();
        const daily = {
          time: dates,
          weather_code: dates.map(d => {
            const codes = dailyMap[d].weatherCodes;
            const freq = {};
            for (const c of codes) freq[c] = (freq[c] || 0) + 1;
            return parseInt(Object.entries(freq).sort((a, b) => b[1] - a[1])[0][0]);
          }),
          temperature_2m_max: dates.map(d => Math.round(Math.max(...dailyMap[d].temps) * 10) / 10),
          temperature_2m_min: dates.map(d => Math.round(Math.min(...dailyMap[d].temps) * 10) / 10),
          precipitation_sum: dates.map(d => Math.round(dailyMap[d].precip * 100) / 100),
          precipitation_probability_max: dates.map(() => 0),
          wind_speed_10m_max: dates.map(d => Math.round(Math.max(...dailyMap[d].windSpeeds) * 10) / 10),
          wind_direction_10m_dominant: dates.map(d => Math.round(dailyMap[d].windDirs.reduce((a, b) => a + b, 0) / dailyMap[d].windDirs.length))
        };
        
        const current = {
          temperature_2m: currentData.main.temp,
          relative_humidity_2m: currentData.main.humidity,
          apparent_temperature: currentData.main.feels_like,
          precipitation: currentData.rain ? (currentData.rain['1h'] || currentData.rain['3h'] || 0) : 0,
          weather_code: currentData.weather[0].id,
          wind_speed_10m: currentData.wind.speed,
          wind_direction_10m: currentData.wind.deg,
          surface_pressure: currentData.main.pressure,
          visibility: currentData.visibility
        };
        
        return { current, daily, hourly: hourlyData };
      } catch (e) {
        console.error('获取天气数据失败:', city.name, e);
        return null;
      }
    }

    // 获取空气质量数据（和风天气）
    async function fetchAirQuality(city) {
      try {
        const url = `https://${QWEATHER_API_HOST}/airquality/v1/current/${city.lat.toFixed(2)}/${city.lon.toFixed(2)}`;
        const res = await fetch(url, {
          headers: { 'X-QW-Api-Key': QWEATHER_API_KEY }
        });
        const data = await res.json();
        if (data && data.pollutants) {
          let pm25 = null, pm10 = null;
          data.pollutants.forEach(p => {
            if (p.code === 'pm2p5') pm25 = p.concentration.value;
            if (p.code === 'pm10') pm10 = p.concentration.value;
          });
          // 获取AQI等级
          let aqi = null, category = '';
          if (data.indexes && data.indexes.length > 0) {
            const idx = data.indexes[0];
            aqi = idx.aqi;
            category = idx.category;
          }
          return { current: { pm2_5: pm25, pm10: pm10, aqi: aqi, category: category }, daily: { pm2_5: [pm25], pm10: [pm10] } };
        }
        return { current: null, daily: null };
      } catch (e) {
        console.error('获取空气质量失败:', city.name, e);
        return { current: null, daily: null };
      }
    }

    // 获取空气质量每日预报（和风天气）
    async function fetchAirQualityDaily(city) {
      try {
        const url = `https://${QWEATHER_API_HOST}/airquality/v1/daily/${city.lat.toFixed(2)}/${city.lon.toFixed(2)}`;
        console.log(`[AQI] 请求预报: ${url}`);
        const res = await fetch(url, {
          headers: { 'X-QW-Api-Key': QWEATHER_API_KEY }
        });
        const data = await res.json();
        console.log(`[AQI] 预报响应:`, data);
        if (data && data.days) {
          const daily = data.days.map(d => {
            let pm25 = null, pm10 = null;
            if (d.pollutants) {
              d.pollutants.forEach(p => {
                if (p.code === 'pm2p5') pm25 = p.concentration.value;
                if (p.code === 'pm10') pm10 = p.concentration.value;
              });
            }
            return { pm2_5: pm25, pm10: pm10 };
          });
          return { daily: { pm2_5: daily.map(d => d.pm2_5), pm10: daily.map(d => d.pm10) } };
        }
        return null;
      } catch (e) {
        console.error('[AQI] 预报请求失败:', city.name, e);
        return null;
      }
    }

    // 渲染天气标记
    async function renderWeatherMarkers() {
      if (!window.weatherLayer) {
        window.weatherLayer = L.layerGroup().addTo(window.weatherMap);
      }
      window.weatherLayer.clearLayers();
      
      for (const city of CITIES) {
        // 使用缓存的数据
        let weatherData = window.cachedWeatherData[city.name];
        let airData = null;
        if (!weatherData) {
          weatherData = await fetchWeatherData(city);
          airData = await fetchAirQuality(city);
          // 额外获取和风天气3天空气质量预报
          try {
            const dailyAir = await fetchAirQualityDaily(city);
            if (dailyAir && dailyAir.daily && dailyAir.daily.pm2_5 && dailyAir.daily.pm2_5.length > 0) {
              airData = { ...airData, daily: dailyAir.daily };
              console.log(`[AQI] ${city.name} 3天预报:`, dailyAir.daily.pm2_5);
            } else {
              console.warn(`[AQI] ${city.name} 预报数据为空`);
            }
          } catch(e) {
            console.warn('空气质量预报获取失败:', city.name, e);
          }
          if (weatherData) window.cachedWeatherData[city.name] = { weatherData, airData };
        } else {
          airData = weatherData.airData;
          weatherData = weatherData.weatherData;
        }
        
        if (weatherData && weatherData.current) {
          const current = weatherData.current;
          const daily = weatherData.daily;
          const weatherCode = current.weather_code;
          const icon = getWeatherIcon(weatherCode);
          
          let highTemp = '--', lowTemp = '--';
          if (daily && daily.temperature_2m_max && daily.temperature_2m_max[0]) {
            highTemp = Math.round(daily.temperature_2m_max[0]);
            lowTemp = Math.round(daily.temperature_2m_min[0]);
          }
          
          const labelHtml = `
            <div class="weather-label">
              <span class="wl-city">${city.name}</span>
              <span class="wl-icon">${icon}</span>
              <span class="wl-temps">
                <span class="wl-high">${highTemp}°</span>
                <span class="wl-sep">/</span>
                <span class="wl-low">${lowTemp}°</span>
              </span>
            </div>
          `;
          
          const marker = L.marker([city.lat, city.lon], {
            icon: L.divIcon({
              html: labelHtml,
              className: '',
              iconSize: [0, 0],
              iconAnchor: [0, 0]
            })
          }).addTo(window.weatherLayer);
          
          marker.on('click', function() {
            showCityForecastPopup(city, weatherData, airData, window.cachedWarningData[city.name] || null);
          });
        }
      }
    }

    // 渲染台风路径
    async function renderTyphoonTracks() {
      if (!window.typhoonLayer) {
        window.typhoonLayer = L.layerGroup().addTo(window.weatherMap);
      }
      window.typhoonLayer.clearLayers();
      
      try {
        let typhoonList = window.cachedTyphoonData;
        if (typhoonList.length === 0) {
          typhoonList = await fetchTyphoonList();
          window.cachedTyphoonData = typhoonList;
        }
        
        for (const tc of typhoonList) {
          const trackData = await fetchTyphoonTrack(tc.id);
          if (trackData && trackData.track && trackData.track.length > 1) {
            const latlngs = trackData.track.map(p => [parseFloat(p.lat), parseFloat(p.lon)]);
            
            // 绘制台风路径线（实线，历史轨迹）
            L.polyline(latlngs, {
              color: '#e74c3c',
              weight: 3,
              opacity: 0.8
            }).addTo(window.typhoonLayer);
            
            // 绘制台风路径点 - 按强度着色
            trackData.track.forEach(p => {
              const ws = parseFloat(p.windSpeed) || 0;
              const color = getTyphoonIntensityColor(ws);
              const intensityLabel = getTyphoonIntensityLabel(ws);
              L.circleMarker([parseFloat(p.lat), parseFloat(p.lon)], {
                radius: ws >= 51 ? 7 : ws >= 32.7 ? 5 : ws >= 24.5 ? 4 : 3,
                color: color,
                fillColor: color,
                fillOpacity: 0.8,
                weight: 1.5
              }).bindTooltip(`${tc.name}\n${intensityLabel}\n风速: ${ws} m/s\n气压: ${p.pressure} hPa\n时间: ${p.time}`, {
                direction: 'top'
              }).addTo(window.typhoonLayer);
            });
            
            // 标注台风名称和当前强度 - 无红色方框，仅台风图标
            if (trackData.track.length > 0) {
              const latest = trackData.track[trackData.track.length - 1];
              const ws = parseFloat(latest.windSpeed) || 0;
              const intensityLabel = getTyphoonIntensityLabel(ws);
              const stormId = tc.id;
              
              const typhoonIcon = L.marker([parseFloat(latest.lat), parseFloat(latest.lon)], {
                icon: L.divIcon({
                  html: `<div style="font-size:32px;line-height:1;text-shadow:0 0 10px rgba(0,0,0,0.5);cursor:pointer;filter:drop-shadow(0 2px 4px rgba(0,0,0,0.3));" class="typhoon-click" data-stormid="${stormId}">🌀</div>`,
                  className: '',
                  iconSize: [36, 36],
                  iconAnchor: [18, 18]
                })
              }).addTo(window.typhoonLayer);
              
              typhoonIcon.on('click', function() {
                showTyphoonDetail(stormId);
              });
              
              // 台风名称标签
              L.marker([parseFloat(latest.lat) + 1.5, parseFloat(latest.lon)], {
                icon: L.divIcon({
                  html: `<div style="color:#fff;font-size:13px;font-weight:bold;text-shadow:0 0 6px rgba(0,0,0,0.9),0 0 12px rgba(0,0,0,0.6);white-space:nowrap;text-align:center;">${tc.name}<br>${intensityLabel} ${ws}m/s</div>`,
                  className: '',
                  iconSize: [0, 0],
                  iconAnchor: [0, 0]
                })
              }).addTo(window.typhoonLayer);
              
              // 绘制风圈（7级/10级/12级）
              const lat = parseFloat(latest.lat);
              const lon = parseFloat(latest.lon);
              if (latest.windRadius30) {
                const ne = parseFloat(latest.windRadius30.neRadius) || 0;
                const se = parseFloat(latest.windRadius30.seRadius) || 0;
                const sw = parseFloat(latest.windRadius30.swRadius) || 0;
                const nw = parseFloat(latest.windRadius30.nwRadius) || 0;
                const avg30 = (ne + se + sw + nw) / 4;
                if (avg30 > 0) drawWindCircle(window.typhoonLayer, lat, lon, avg30, '#f1c40f', '7级风圈');
              }
              if (latest.windRadius50) {
                const ne = parseFloat(latest.windRadius50.neRadius) || 0;
                const se = parseFloat(latest.windRadius50.seRadius) || 0;
                const sw = parseFloat(latest.windRadius50.swRadius) || 0;
                const nw = parseFloat(latest.windRadius50.nwRadius) || 0;
                const avg50 = (ne + se + sw + nw) / 4;
                if (avg50 > 0) drawWindCircle(window.typhoonLayer, lat, lon, avg50, '#e67e22', '10级风圈');
              }
              if (latest.windRadius64) {
                const ne = parseFloat(latest.windRadius64.neRadius) || 0;
                const se = parseFloat(latest.windRadius64.seRadius) || 0;
                const sw = parseFloat(latest.windRadius64.swRadius) || 0;
                const nw = parseFloat(latest.windRadius64.nwRadius) || 0;
                const avg64 = (ne + se + sw + nw) / 4;
                if (avg64 > 0) drawWindCircle(window.typhoonLayer, lat, lon, avg64, 'rgba(231,76,60,0.25)', '12级风圈');
              }
              
              // 保存当前台风位置供用户预报使用
              window.currentTyphoonPosition = {
                lat: parseFloat(latest.lat),
                lon: parseFloat(latest.lon),
                color: getTyphoonIntensityColor(parseFloat(latest.windSpeed) || 0)
              };
              
              // 获取并绘制预报路径（受预测路径开关控制）
              if (window.showForecastPath) {
                try {
                  const forecastData = await fetchTyphoonForecast(tc.id);
                  if (forecastData && forecastData.length > 0) {
                    // 获取当前台风位置（历史轨迹最后一个点）
                    const currentLat = parseFloat(latest.lat);
                    const currentLon = parseFloat(latest.lon);
                    const currentWindSpeed = parseFloat(latest.windSpeed) || 0;
                    const currentColor = getTyphoonIntensityColor(currentWindSpeed);
                    
                    // 预报路径包含当前台风位置作为起点
                    const forecastLatlngs = [[currentLat, currentLon], ...forecastData.map(p => [parseFloat(p.lat), parseFloat(p.lon)])];
                    
                    // 绘制预报路径（虚线），第一段使用当前台风强度颜色
                    L.polyline(forecastLatlngs, {
                      color: currentColor,
                      weight: 2,
                      opacity: 0.6,
                      dashArray: '8, 6'
                    }).addTo(window.typhoonLayer);
                  // 预报点按强度着色
                  forecastData.forEach(p => {
                    const ws = parseFloat(p.windSpeed) || 0;
                    const color = getTyphoonIntensityColor(ws);
                    const intensityLabel = getTyphoonIntensityLabel(ws);
                    L.circleMarker([parseFloat(p.lat), parseFloat(p.lon)], {
                      radius: 5,
                      color: color,
                      fillColor: '#fff',
                      fillOpacity: 0.9,
                      weight: 2.5
                    }).bindTooltip(`${tc.name} 预报\n${intensityLabel}\n风速: ${ws} m/s\n气压: ${p.pressure} hPa\n时间: ${p.fxTime}`, {
                      direction: 'top'
                    }).addTo(window.typhoonLayer);
                  });
                }
              } catch(e) { console.error('获取台风预报失败:', e); }
              }  // end if showForecastPath
            }
          }
        }
      } catch (e) {
        console.error('获取台风路径失败:', e);
      }
    }

    // 切换按钮点击处理
    function setupToggleButtons() {
      const buttons = document.querySelectorAll('.map-toggle-btn');
      buttons.forEach(btn => {
        btn.addEventListener('click', async function() {
          const toggle = this.dataset.toggle;
          const isActive = window.toggleState[toggle];
          
          // 切换状态
          window.toggleState[toggle] = !isActive;
          this.classList.toggle('active');
          this.classList.toggle(`active-${toggle}`);
          
          // 根据切换状态显示/隐藏图层
          if (toggle === 'forecast') {
            if (window.toggleState.forecast) {
              if (window.weatherLayer) window.weatherMap.addLayer(window.weatherLayer);
              // 如果预警也开启，添加预警图层
              if (window.toggleState.warning && window.warningLayer) {
                window.weatherMap.addLayer(window.warningLayer);
              }
            } else {
              if (window.weatherLayer) window.weatherMap.removeLayer(window.weatherLayer);
              // 如果预警单独开启，只显示预警
              if (window.toggleState.warning && window.warningLayer) {
                window.weatherMap.addLayer(window.warningLayer);
              } else if (window.warningLayer) {
                window.weatherMap.removeLayer(window.warningLayer);
              }
            }
          } else if (toggle === 'warning') {
            if (window.toggleState.warning) {
              if (window.warningLayer) window.weatherMap.addLayer(window.warningLayer);
            } else {
              if (window.warningLayer) window.weatherMap.removeLayer(window.warningLayer);
            }
          } else if (toggle === 'typhoon') {
            if (window.toggleState.typhoon) {
              if (!window.typhoonDataLoaded) {
                window.typhoonDataLoaded = true;
                await renderTyphoonTracks();
              } else if (window.typhoonLayer) {
                window.weatherMap.addLayer(window.typhoonLayer);
              }
            } else {
              if (window.typhoonLayer) window.weatherMap.removeLayer(window.typhoonLayer);
            }
          }
        });
      });
    }

        else if (ws < 51.0) color = '#e91e63'; // 强台风 粉
        else color = '#e74c3c';                 // 超强台风 红
        
        const point = {
          lat: parseFloat(lat),
          lon: parseFloat(lon),
          period: period,
          level: level,
          windSpeed: ws,
          color: color
        };
        
        window.userForecastData.push(point);
        window.userForecastStep++;

            currentTyphoon = window.typhoonPositions[selectedName];
          } else {
            currentTyphoon = window.currentTyphoonPosition || null;
          }
          if (currentTyphoon) {
            L.polyline([
              [currentTyphoon.lat, currentTyphoon.lon],
              [point.lat, point.lon]
            ], {
              color: currentTyphoon.color,  // 使用当前台风强度的颜色
              weight: 2,
              opacity: 0.7,
              dashArray: '6, 4'
            }).addTo(window.userForecastLayer);
          }
        } else if (window.userForecastData.length > 1) {
          // 后续预报点：与上一个点连线
          const prev = window.userForecastData[window.userForecastData.length - 2];
          L.polyline([
            [prev.lat, prev.lon],
            [point.lat, point.lon]
          ], {
            color: prev.color,
            weight: 2,
            opacity: 0.7,
            dashArray: '6, 4'
          }).addTo(window.userForecastLayer);
        }
        
        // 标注点
        L.circleMarker([point.lat, point.lon], {
          radius: 6,
          color: color,
          fillColor: '#fff',
          fillOpacity: 0.9,
          weight: 3
        }).bindTooltip(`✏️ 个人预报\n时段: ${period}h\n等级: ${level}\n风速: ${ws} m/s`, {
          direction: 'top'
        }).addTo(window.userForecastLayer);
        
        // 时段标签
        L.marker([point.lat + 0.5, point.lon], {
          icon: L.divIcon({
            html: `<div style="color:${color};font-size:11px;font-weight:bold;text-shadow:0 0 6px rgba(0,0,0,0.8);white-space:nowrap;background:rgba(0,0,0,0.5);padding:2px 6px;border-radius:4px;">${period}h</div>`,
            className: '',
            iconSize: [0, 0]
          })
        }).addTo(window.userForecastLayer);
        
        if (window.userForecastStep >= FORECAST_PERIODS.length) {
          alert('所有时段预报点已添加完毕！\n路线将在5分钟后自动清除。');

          setTimeout(() => {
            if (window.userForecastLayer) {
              window.userForecastLayer.clearLayers();
              window.userForecastData = [];
              window.userForecastStep = 0;
              window.userForecastMode = false;
              document.getElementById('userForecastBtn')?.classList.remove('active');
              window.weatherMap.getContainer().style.cursor = '';
            }
          }, 5 * 60 * 1000);
        }
      });
    }

      if (window.weatherMapInitialized) return;

        preferCanvas: true
      }).setView([35, 110], 4);

        attribution: '&copy; 天地图',
        subdomains: ['0', '1', '2', '3', '4', '5', '6', '7'],
        maxZoom: 18
      }).addTo(window.weatherMap);
      
      // 天地图中文注记
      L.tileLayer('https://t{s}.tianditu.gov.cn/cva_w/wmts?SERVICE=WMTS&REQUEST=GetTile&VERSION=1.0.0&LAYER=cva&STYLE=default&TILEMATRIXSET=w&FORMAT=tiles&TILEMATRIX={z}&TILEROW={y}&TILECOL={x}&tk=' + TIANDITU_TK, {
        subdomains: ['0', '1', '2', '3', '4', '5', '6', '7'],
        maxZoom: 18
      }).addTo(window.weatherMap);

      // 创建图层组
      window.weatherLayer = L.layerGroup().addTo(window.weatherMap);
      window.warningLayer = L.layerGroup().addTo(window.weatherMap);
      window.typhoonLayer = L.layerGroup();
      window.typhoonDataLoaded = false;

      // 并行加载天气和预警数据
      await Promise.all([
        renderWeatherMarkers(),
        renderWarningMarkers()
      ]);

      // 如果台风按钮默认开启，加载台风数据
      if (window.toggleState.typhoon) {
        window.typhoonDataLoaded = true;
        renderTyphoonTracks();
      }

      // 设置切换按钮
      setupToggleButtons();
      setupForecastPathToggle();
      setupUserForecast();
      
      window.weatherMapInitialized = true;
    }

        if (p25 !== null) pm25Level = p25 <= 35 ? '优' : p25 <= 75 ? '良' : p25 <= 115 ? '轻度' : p25 <= 150 ? '中度' : '重度';
        if (p10 !== null) pm10Level = p10 <= 50 ? '优' : p10 <= 150 ? '良' : p10 <= 250 ? '轻度' : p10 <= 350 ? '中度' : '重度';
      }
      
      const dayNames = ['周日', '周一', '周二', '周三', '周四', '周五', '周六'];
      const windDirsShort = ['北', '东北', '东', '东南', '南', '西南', '西', '西北'];
      
      function getDailyAvg(hourlyData, field, dayIndex) {
        if (!hourlyData || !hourlyData[field]) return '—';

        if (i === 0) {

        } else if (i >= 1 && i <= 3) {

          }

        // i >= 4 时不显示AQI，保持"—"
        
        daysHtml += `
          <div class="forecast-card">
            <div class="fc-header">
              <span class="fc-day">${dayName}</span>
              <span class="fc-icon">${dayIcon}</span>
              <span class="fc-desc">${dayDesc}</span>
              <span class="fc-temps">
                <span class="fc-high">${high}°</span>
                <span class="fc-sep">/</span>
                <span class="fc-low">${low}°</span>
              </span>
            </div>
            <div class="fc-body">
              <div class="fc-row">
                <span class="fc-item">💧 湿度 ${avgHumidity}</span>
                <span class="fc-item">💨 ${windDirText}风 ${wind}km/h</span>
                <span class="fc-item">🌧️ ${rain.toFixed(1)}mm ${getRainLevel(rain)}</span>
              </div>
              <div class="fc-row">
                <span class="fc-item">🔽 气压 ${avgPressure}</span>
                <span class="fc-item">👁️ 能见度 ${avgVis}</span>
                <span class="fc-item">🌬️ ${aqiText}</span>
              </div>
            </div>
          </div>
        `;
      }
      
      let warningHtml = '';
      if (warning) {
        const warnColor = WARNING_LEVEL_COLORS[warning.color?.code] || WARNING_LEVEL_COLORS[warning.severity] || 'rgba(234,67,53,0.9)';
        warningHtml = `
          <div class="popup-warning" style="background:${warnColor}20;border-left:4px solid ${warnColor};padding:8px 12px;margin:-8px -12px 10px -12px;border-radius:4px;">
            <div style="font-weight:bold;color:${warnColor};font-size:14px;">⚠️ ${warning.eventType?.name || ''} ${warning.severity || ''}预警</div>
            <div style="font-size:12px;color:#555;margin-top:4px;">${warning.headline || ''}</div>
            <div style="font-size:11px;color:#777;margin-top:2px;line-height:1.5;">${warning.description ? warning.description.slice(0, 200) + (warning.description.length > 200 ? '...' : '') : ''}</div>
          </div>
        `;
      }
      
      const popupHtml = `
        <div class="forecast-popup">
          <div class="popup-header">
            <h3>${icon} ${city.name}</h3>
            <div class="sub-info">${desc} · 体感 ${feelsLike}°C</div>
          </div>
          ${warningHtml}
          <div class="popup-body">
            <div class="popup-section">
              <div class="popup-grid">
                <div class="popup-grid-item">
                  <div class="label">🌡️ 气温</div>
                  <div class="value warm">${temp}°C</div>
                </div>
                <div class="popup-grid-item">
                  <div class="label">💧 湿度</div>
                  <div class="value">${humidity}%</div>
                </div>
                <div class="popup-grid-item">
                  <div class="label">💨 风速</div>
                  <div class="value wind">${windSpeed} km/h</div>
                  <div style="font-size:10px;color:#999">${windDirText}风</div>
                </div>
                <div class="popup-grid-item">
                  <div class="label">🌧️ 降水量</div>
                  <div class="value">${precip} mm ${getRainLevel(precip)}</div>
                </div>
                <div class="popup-grid-item">
                  <div class="label">🔽 气压</div>
                  <div class="value">${Math.round(pressure)} hPa</div>
                </div>
                <div class="popup-grid-item">
                  <div class="label">👁️ 能见度</div>
                  <div class="value">${visibility ? (visibility/1000).toFixed(1) + ' km' : '—'}</div>
                </div>
              </div>
            </div>
            <div class="popup-section">
              <div class="popup-section-title">🌬️ 空气质量</div>
              <div class="aqi-section">
                <div class="aqi-row">
                  <div class="aqi-item">
                    <div class="aqi-label">PM2.5</div>
                    <div class="aqi-value">${pm25}</div>
                    <div><span class="aqi-badge" style="background:${pm25Level === '优' ? '#27ae60' : pm25Level === '良' ? '#f39c12' : '#e74c3c'}">${pm25Level}</span></div>
                  </div>
                  <div class="aqi-item">
                    <div class="aqi-label">PM10</div>
                    <div class="aqi-value">${pm10}</div>
                    <div><span class="aqi-badge" style="background:${pm10Level === '优' ? '#27ae60' : pm10Level === '良' ? '#f39c12' : '#e74c3c'}">${pm10Level}</span></div>
                  </div>
                </div>
              </div>
            </div>
            <div class="popup-section">
              <div class="popup-section-title">📅 未来7天天气预报</div>
              ${daysHtml}
            </div>
            <div class="popup-section">
              <div class="popup-section-title">📈 气象要素趋势图</div>
              <div class="chart-container">
                <canvas id="forecastChart"></canvas>
              </div>
            </div>
          </div>
        </div>
      `;
      
      L.popup({ maxWidth: 380, minWidth: 320, className: 'weather-popup' })
        .setLatLng([city.lat, city.lon])
        .setContent(popupHtml)
        .openOn(window.weatherMap);
      
      setTimeout(() => {
        const canvas = document.getElementById('forecastChart');
        if (!canvas) return;
        
        const labels = [];
        const highTemps = [];
        const lowTemps = [];
        const pressures = [];
        const windSpeeds = [];
        const rains = [];
        
        for (let i = 0; i < Math.min(7, daily.time.length); i++) {
          const date = new Date(daily.time[i] + 'T12:00:00');
          const dayName = i === 0 ? '今天' : i === 1 ? '明天' : (date.getMonth()+1)+'/'+date.getDate();
          labels.push(dayName);
          highTemps.push(Math.round(daily.temperature_2m_max[i]));
          lowTemps.push(Math.round(daily.temperature_2m_min[i]));
          const press = getDailyAvg(weatherData.hourly, 'surface_pressure', i);
          pressures.push(press !== '—' ? parseInt(press) : null);
          const wind = daily.wind_speed_10m_max ? Math.round(daily.wind_speed_10m_max[i]) : 0;
          windSpeeds.push(wind);
          const rain = daily.precipitation_sum ? daily.precipitation_sum[i] : 0;
          rains.push(rain);
        }
        
        if (window._forecastChart) window._forecastChart.destroy();
        
        window._forecastChart = new Chart(canvas, {

              { label: '最低温(°C)', data: lowTemps, borderColor: '#3498db', backgroundColor: 'rgba(52,152,219,0.1)', tension: 0.3, pointRadius: 4, pointBackgroundColor: '#3498db', borderWidth: 2, fill: false, yAxisID: 'y' },

    const API_BASE_URL = 'http://localhost:5000'; // API 服务器地址
    
    let currentTaskId = null;
    let uploadedFile = null;
    
    // 上传区域交互
    const uploadArea = document.getElementById('ta-upload-area');
    const fileInput = document.getElementById('ta-file-input');
    const fileInfo = document.getElementById('ta-file-info');
    const fileName = document.getElementById('ta-file-name');
    const clearFileBtn = document.getElementById('ta-clear-file');
    const generateBtn = document.getElementById('ta-generate-btn');
    
    uploadArea.addEventListener('click', () => fileInput.click());
    
    uploadArea.addEventListener('dragover', (e) => {
      e.preventDefault();
      uploadArea.classList.add('dragover');
    });
    
    uploadArea.addEventListener('dragleave', () => {
      uploadArea.classList.remove('dragover');
    });
    
    uploadArea.addEventListener('drop', (e) => {
      e.preventDefault();
      uploadArea.classList.remove('dragover');
      const files = e.dataTransfer.files;
      if (files.length > 0) {
        handleFileUpload(files[0]);
      }
    });
    
    fileInput.addEventListener('change', (e) => {
      if (e.target.files.length > 0) {
        handleFileUpload(e.target.files[0]);
      }
    });
    
    clearFileBtn.addEventListener('click', () => {
      uploadedFile = null;
      fileInput.value = '';
      fileInfo.style.display = 'none';
      uploadArea.style.display = 'block';
      generateBtn.disabled = true;
    });
    
    async function handleFileUpload(file) {
      if (!file.name.endsWith('.dat')) {
        alert('请上传 .dat 格式的 ATCF 文件');
        return;
      }
      
      uploadedFile = file;
      fileName.textContent = file.name;
      fileInfo.style.display = 'flex';
      uploadArea.style.display = 'none';
      
      // 上传文件到服务器
      try {
        const formData = new FormData();
        formData.append('file', file);
        
        const response = await fetch(`${API_BASE_URL}/api/upload`, {
          method: 'POST',
          body: formData
        });
        
        const data = await response.json();
        
        if (response.ok) {
          currentTaskId = data.task_id;
          generateBtn.disabled = false;
        } else {
          alert('上传失败：' + data.error);
          uploadedFile = null;
          fileInfo.style.display = 'none';
          uploadArea.style.display = 'block';
        }
      } catch (error) {
        alert('上传失败：' + error.message);
        console.error('上传错误:', error);
      }
    }
    
    // 生成按钮
    generateBtn.addEventListener('click', async () => {
      if (!currentTaskId) {
        alert('请先上传文件');
        return;
      }
      
      generateBtn.disabled = true;
      document.getElementById('ta-progress-section').style.display = 'block';
      document.getElementById('ta-result-section').style.display = 'none';
      document.getElementById('ta-error-section').style.display = 'none';
      
      try {
        const response = await fetch(`${API_BASE_URL}/api/generate`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ task_id: currentTaskId })
        });
        
        const data = await response.json();
        
        if (response.ok && data.status === 'completed') {
          document.getElementById('ta-progress-fill').style.width = '100%';
          document.getElementById('ta-progress-text').textContent = '生成完成！';
          document.getElementById('ta-result-section').style.display = 'block';
          document.getElementById('ta-result-message').textContent = '台风动画已成功生成！';
          document.getElementById('ta-download-btn').href = `${API_BASE_URL}/api/download/${currentTaskId}`;
        } else {
          document.getElementById('ta-error-section').style.display = 'block';
          document.getElementById('ta-error-message').textContent = data.error || '生成失败';
        }
      } catch (error) {
        document.getElementById('ta-error-section').style.display = 'block';
        document.getElementById('ta-error-message').textContent = '请求失败：' + error.message;
      } finally {
        generateBtn.disabled = false;
      }
    });
    
    // 轮询任务状态（用于长时间任务）
    async function pollTaskStatus(taskId) {
      try {
        const response = await fetch(`${API_BASE_URL}/api/status/${taskId}`);
        const data = await response.json();
        
        if (data.status === 'processing') {
          document.getElementById('ta-progress-text').textContent = data.message || '处理中...';
          setTimeout(() => pollTaskStatus(taskId), 2000);
        } else if (data.status === 'completed') {
          document.getElementById('ta-progress-fill').style.width = '100%';
          document.getElementById('ta-progress-text').textContent = '生成完成！';
          document.getElementById('ta-result-section').style.display = 'block';
          document.getElementById('ta-result-message').textContent = '台风动画已成功生成！';
          document.getElementById('ta-download-btn').href = `${API_BASE_URL}/api/download/${taskId}`;
          generateBtn.disabled = false;
        } else if (data.status === 'error') {
          document.getElementById('ta-error-section').style.display = 'block';
          document.getElementById('ta-error-message').textContent = data.message || '生成失败';
          generateBtn.disabled = false;
        }
      } catch (error) {
        console.error('状态查询失败:', error);
      }
    }
